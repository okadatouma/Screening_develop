"""[物理] 解析式から局所エネルギーを検算。旧実装との一致とは区別する。"""
import unittest

from tests.support import require_numerics
require_numerics()

import jax
import jax.numpy as jnp
import numpy as np
from periodicwave import hamiltonian, networks
from periodicwave.pbc import hamiltonian as PH, lambda_jellium as LJ


def one_particle(x=(0.31, -0.27), spin=1.):
    return networks.FermiNetData(positions=jnp.asarray(x), spins=jnp.asarray([spin]),
                               atoms=jnp.zeros((1, 2)), charges=jnp.zeros(1))


def complex_gaussian(p, x, spins, atoms, charges):
    # ψ = exp[(-a+ib)|x|²/2 + i k.x]。
    return 0.3 * jnp.sum(x*x) / 2 + jnp.dot(jnp.asarray([0.7, -0.2]), x), -0.8 * jnp.sum(x*x) / 2


class IndependentPhysics(unittest.TestCase):
    def test_complex_kinetic_including_imaginary_part(self):
        d = one_particle()
        z = -0.8 + 0.3j
        g = z * d.positions + 1j * jnp.asarray([0.7, -0.2])
        expected = -0.5 * (2*z + jnp.sum(g*g))  # 共役を入れない。
        self.assertGreater(abs(complex(expected).imag), 0.01)
        for method in ("default", "folx"):
            for scan in (False, True):
                with self.subTest(method=method, scan=scan):
                    kinetic = hamiltonian.local_kinetic_energy(
                        complex_gaussian, use_scan=scan, complex_output=True, laplacian_method=method)
                    np.testing.assert_allclose(kinetic(None, d), expected, rtol=1e-10, atol=1e-10)

    def test_plane_wave_kinetic(self):
        q = jnp.asarray([0.7, -0.2])
        f = lambda p, x, s, a, c: (q @ x, jnp.asarray(0.))
        for method in ("default", "folx"):
            kinetic = hamiltonian.local_kinetic_energy(f, complex_output=True, laplacian_method=method)
            np.testing.assert_allclose(kinetic(None, one_particle()), jnp.dot(q, q) / 2, atol=1e-12)

    def test_lambda_lower_upper_band_both_pseudospins(self):
        q = jnp.asarray([0.7, -0.2])
        lam, delta = 0.4, 2.
        q2 = float(q @ q)
        for upper in (False, True):
            u = (jnp.asarray([-lam * (q[0] - 1j*q[1]), 1.+0j]) if upper
                 else jnp.asarray([1.+0j, lam * (q[0] + 1j*q[1])]))
            def f(p, x, s, a, c):
                value = jnp.where(s[0] > 0, u[0], u[1]) * jnp.exp(1j * (q @ x))
                return jnp.angle(value), jnp.log(jnp.abs(value))
            expected = q2/2 + (delta * (1 + lam**2 * q2) if upper else 0.)
            for method in ("default", "folx"):
                el = LJ.local_energy(f, jnp.zeros(1), (1, 0), lattice=jnp.eye(2)*4,
                                     lam=lam, delta=delta, potential_type="none",
                                     kinetic_kwargs={"laplacian_method": method})
                for spin in (-1., 1.):
                    with self.subTest(upper=upper, method=method, spin=spin):
                        np.testing.assert_allclose(el(None, None, one_particle(spin=spin))[0], expected, atol=1e-10)

    def test_lambda_zero_and_component_sum(self):
        d = one_particle()
        ordinary = hamiltonian.local_kinetic_energy(complex_gaussian, complex_output=True)
        for spin in (-1., 1.):
            dd = one_particle(spin=spin)
            local = LJ.local_energy(complex_gaussian, d.charges, (1, 0), lattice=jnp.eye(2)*4,
                                    lam=0., delta=2., potential_type="none",
                                    kinetic_kwargs={"laplacian_method": "default"})
            expected = ordinary(None, dd) + (2. if spin < 0 else 0.)
            np.testing.assert_allclose(local(None, None, dd)[0], expected, atol=1e-10)
        local = LJ.local_energy(complex_gaussian, d.charges, (1, 0), lattice=jnp.eye(2)*4,
                                lam=0.4, delta=2., potential_type="none")
        components = LJ.make_energy_components(complex_gaussian, jnp.eye(2)*4, 0.4, 2., 1., coulomb=False)
        np.testing.assert_allclose(components(None, d).sum(), local(None, None, d)[0], atol=1e-10)

    def test_ewald_lattice_translation_and_scaling(self):
        lattice = jnp.asarray([[3.2, -1.6], [0., 2.7712812921102037]])
        displacement = jnp.asarray([0.47, 0.63])
        ee = jnp.stack([jnp.stack([jnp.zeros(2), displacement]),
                        jnp.stack([-displacement, jnp.zeros(2)])])
        potential = PH.make_2DCoulomb_potential(lattice, jnp.zeros(1), jnp.zeros(1), 3, 1.)
        stronger = PH.make_2DCoulomb_potential(lattice, jnp.zeros(1), jnp.zeros(1), 3, 2.)
        shift = lattice[:, 1]
        shifted = ee.at[0, 1].add(shift).at[1, 0].add(-shift)
        np.testing.assert_allclose(potential(None, ee), potential(None, shifted), atol=1e-10)
        np.testing.assert_allclose(stronger(None, ee), 2 * potential(None, ee), atol=1e-10)


class KnownEnergyDifference(unittest.TestCase):
    def energies(self):
        options = dict(f=complex_gaussian, charges=jnp.zeros(1), nspins=(1, 0),
                       complex_output=True, lattice=jnp.eye(2)*4, convergence_radius=2,
                       potential_kwargs={"interaction_energy_scale": 1.})
        return [PH.local_energy(**options, kinetic_kwargs={"laplacian_method": method})(None, None, one_particle())[0]
                for method in ("default", "folx")]

    def test_current_folx_path_discards_only_imaginary_part(self):
        default, forward = self.energies()
        self.assertGreater(abs(complex(default).imag), 0.01)
        np.testing.assert_allclose(jnp.real(default), forward, atol=1e-10)
        self.assertEqual(float(jnp.imag(forward)), 0.)

    @unittest.expectedFailure
    def test_K03_both_paths_should_preserve_complex_local_energy(self):
        default, forward = self.energies()
        np.testing.assert_allclose(default, forward, atol=1e-10)

    def test_K05_NoneMoire_is_currently_unimplemented(self):
        with self.assertRaises(UnboundLocalError):
            PH.local_energy(complex_gaussian, jnp.zeros(1), (1, 0), complex_output=True,
                            lattice=jnp.eye(2), potential_type="NoneMoire",
                            kinetic_kwargs={"laplacian_method": "default"})
