def stats(a,b):
    a=np.asarray(a);b=np.asarray(b);finite=np.isfinite(a)&np.isfinite(b)
    aa=a[finite].astype(np.float64);bb=b[finite].astype(np.float64);delta=aa-bb
    scale=float(np.max(np.abs(bb))) if bb.size else None
    return dict(shape=list(a.shape),dtype=str(a.dtype),equal_including_nonfinite=bool(np.array_equal(a,b,equal_nan=True)),
        finite_components=int(finite.sum()),a_nonfinite=int((~np.isfinite(a)).sum()),b_nonfinite=int((~np.isfinite(b)).sum()),
        max_abs=float(np.max(np.abs(delta))) if delta.size else None,
        rms=float(np.sqrt(np.mean(delta**2))) if delta.size else None,
        reference_max_abs=scale,relative_to_reference_max=float(np.max(np.abs(delta))/scale) if delta.size and scale else None)

records=[];kernel=[]
for x64 in (False,True):
    jax.config.update('jax_enable_x64',x64)
    for name,L0 in (('square',[[3.2,0.],[0.,3.2]]),('oblique',[[3.2,-1.6],[0.,2.7712812921102037]])):
        L=jnp.asarray(L0);metric=L.T@L
        for label,u in (('regular',[.23,.41]),('near_origin',[1e-8,-2e-8]),('origin',[0.,0.])):
            u=jnp.asarray(u)
            row=dict(x64=x64,lattice=name,point=label)
            for key,transform in (('value',lambda f:f),('gradient',jax.grad),('hessian',jax.hessian)):
                a=transform(lambda r:geometry_norm(metric,r));b=transform(lambda r:FL.periodic_norm(metric,r))
                row[key]=stats(a(u),b(u));row[key+'_jit']=stats(jax.jit(a)(u),jax.jit(b)(u))
                assert row[key]['equal_including_nonfinite'] and row[key+'_jit']['equal_including_nonfinite']
            kernel.append(row)
        geometry=networks.make_periodic_construct_input_features(L)
        feature=FL.make_pbc_feature_layer(1,(2,1),2,lattice=L,include_r_ae=False)
        atoms=jnp.zeros((1,2))
        def geometry_distance(r): return geometry(r,atoms,2)[3][...,0]
        def feature_distance(r):
            ae,ee,ra,re=geometry(r,atoms,2)
            return feature.apply(ae=ae,ee=ee,r_ae=ra,r_ee=re)[1][...,0]
        for label,u in (('regular',[[.13,.21],[.57,.44],[.35,.72]]),

                        ('coincident_distinct_electrons',[[.13,.21],[.13,.21],[.35,.72]])):
            r=(jnp.asarray(u)@L.T).reshape(-1)
            row=dict(x64=x64,lattice=name,configuration=label,positions=np.asarray(r).tolist())
            for key,transform in (('value',lambda f:f),('coordinate_gradient',jax.jacfwd),
                                  ('coordinate_hessian',lambda f:jax.jacfwd(jax.jacrev(f)))):
                a=transform(geometry_distance);b=transform(feature_distance)
                ea,eb=a(r),b(r);ja,jb=jax.jit(a)(r),jax.jit(b)(r)
                row[key]=stats(ea,eb);row[key+'_jit']=stats(ja,jb)
                row[key+'_eager_vs_jit_geometry']=stats(ea,ja)
                row[key+'_eager_vs_jit_feature']=stats(eb,jb)
                row[key+'_diagonal_geometry_nonfinite']=int((~np.isfinite(np.asarray(ja)[np.arange(3),np.arange(3)])).sum())
                row[key+'_diagonal_feature_nonfinite']=int((~np.isfinite(np.asarray(jb)[np.arange(3),np.arange(3)])).sum())
                row[key+'_diagonal_geometry_max_finite_abs']=float(np.max(np.abs(np.nan_to_num(np.asarray(ja)[np.arange(3),np.arange(3)],nan=0.,posinf=0.,neginf=0.))))
                row[key+'_diagonal_feature_max_finite_abs']=float(np.max(np.abs(np.nan_to_num(np.asarray(jb)[np.arange(3),np.arange(3)],nan=0.,posinf=0.,neginf=0.))))
            records.append(row)
            jax.clear_caches()
jax.config.update('jax_enable_x64',True)

# A deterministic search documents why reciprocal conversion order is retained.
roundtrip=[]
generator=np.random.default_rng(41)
lattices=3*np.eye(2)[None]+generator.normal(size=(128,2,2))*.5
for x64 in (False,True):
    jax.config.update('jax_enable_x64',x64)
    inverse=jnp.linalg.inv(jnp.asarray(lattices))
    converted=(2*jnp.pi*inverse)/2/jnp.pi
    difference=np.asarray(converted-inverse)
    found=np.flatnonzero(np.any(difference!=0,axis=(1,2)))
    item=dict(x64=x64,search_lattices=128,changed_reciprocal_matrices=int(found.size),
              max_reciprocal_abs_difference=float(np.max(np.abs(difference))))
    if found.size:
        L=jnp.asarray(lattices[found[0]])
        geometry=networks.make_periodic_construct_input_features(L)
        feature=FL.make_pbc_feature_layer(1,(2,1),2,lattice=L,include_r_ae=False)
        atoms=jnp.zeros((1,2));r=jnp.asarray([.23,.41,1.17,.89,.65,1.4])
        def gd(r):return geometry(r,atoms,2)[3][...,0]
        def fd(r):
            ae,ee,ra,re=geometry(r,atoms,2)
            return feature.apply(ae=ae,ee=ee,r_ae=ra,r_ee=re)[1][...,0]
        item.update(lattice=np.asarray(L).tolist(),positions=np.asarray(r).tolist())
        for key,transform in (('value',lambda f:f),('gradient',jax.jacfwd),('hessian',lambda f:jax.jacfwd(jax.jacrev(f)))):
            a=transform(gd);b=transform(fd)
            item[key]=stats(a(r),b(r));item[key+'_jit']=stats(jax.jit(a)(r),jax.jit(b)(r))
    roundtrip.append(item)
jax.config.update('jax_enable_x64',True)
with (out/'reciprocal-roundtrip.json').open('x') as f:json.dump(roundtrip,f,indent=2,allow_nan=False)
print('Recorded deterministic reciprocal roundtrip search.')
